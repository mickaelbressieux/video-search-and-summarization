# OFFLINE single-file tracking config for the stock `deepstream-app` binary.
#
# Purpose: run RT-DETR detection + NvDCF tracking on a RECORDED file and write
# an OSD-rendered MP4 with bounding boxes + persistent track IDs.
#
# Why this exists (vs run_config-api-rtdetr-protobuf.txt): the API profile feeds
# sources dynamically through `nvmultiurisrcbin` + REST /stream/add. On this box
# that path stalls — the source opens then instantly EOSes with 0 frames
# ("Got data flow before segment event"). Standalone decode is fine (proven:
# 150/150 frames via gst-launch on both HW and SW decoders), so the failure is
# in the nvmultiurisrcbin/new-streammux source path, not the decoder or file.
# This config uses a STATIC [source0] file group + the classic streammux, which
# activates reliably, to get tracked output offline while the REST path is fixed.
#
# Run (inside the vss-rtvi-cv container):
#   cd .../metropolis_perception_app/mounted-configs
#   deepstream-app -c ds_track_bypass.txt
# Input : /opt/storage/input/<file>.mp4   (host: deploy/docker/data-dir/models/input/)
# Output: /opt/storage/output/<file>.mp4  (host: deploy/docker/data-dir/models/output/)
# ---------------------------------------------------------------------------

[application]
enable-perf-measurement=1
perf-measurement-interval-sec=5

[tiled-display]
enable=1
rows=1
columns=1
width=1280
height=720
gpu-id=0
nvbuf-memory-type=0

[source0]
enable=1
# Type - 2=uri 3=MultiURI ; 3 with a single uri = one static file source
type=3
uri=file:///opt/storage/input/dance_clean.mp4
num-sources=1
gpu-id=0
cudadec-memtype=0

[sink0]
enable=1
# Type - 1=FakeSink 2=EglSink 3=File
type=3
# 1=mp4 2=mkv
container=1
# 1=h264 2=h265 3=mpeg4
codec=1
# encoder type 0=Hardware(NVENC, ABSENT on H100) 1=Software(x264enc) -- must be 1 here
enc-type=1
sync=0
bitrate=40000000
output-file=/opt/storage/output/dance_tracked.mp4
source-id=0

[osd]
enable=1
gpu-id=0
border-width=3
text-size=15
text-color=1;1;1;1;
text-bg-color=0.3;0.3;0.3;1
font=Serif
show-clock=0
display-text=1
display-bbox=1
nvbuf-memory-type=0

[streammux]
gpu-id=0
# recorded file, not a live camera
live-source=0
batch-size=1
batched-push-timeout=40000
width=1920
height=1080
enable-padding=1
nvbuf-memory-type=0

[primary-gie]
enable=1
gpu-id=0
batch-size=1
gie-unique-id=1
nvbuf-memory-type=0
# reuse the profile's RT-DETR detector (classes: background,bicycle,car,person,road_sign)
config-file=/opt/nvidia/deepstream/deepstream-9.1/sources/apps/sample_apps/metropolis_perception_app/configs/rtdetr-960x544.txt

[tracker]
enable=1
# tracker-width/height must be multiples of 32
tracker-width=960
tracker-height=544
ll-lib-file=/opt/nvidia/deepstream/deepstream/lib/libnvds_nvmultiobjecttracker.so
# NvDCF accuracy profile (uses cached ReID engine under /opt/engines)
ll-config-file=/opt/nvidia/deepstream/deepstream/samples/configs/deepstream-app/config_tracker_NvDCF_accuracy.yml
gpu-id=0
display-tracking-id=1

[tests]
# single pass: EOS after the file ends so the output MP4 is finalized
file-loop=0
